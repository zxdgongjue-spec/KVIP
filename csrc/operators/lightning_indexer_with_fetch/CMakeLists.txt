cmake_minimum_required(VERSION 3.16)
project(kvip_lifetch LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUN_MODE npu CACHE STRING "NPU build")
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "Target SoC, not runtime device ordinal")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann CACHE PATH "CANN installation")
set(KVAF_INCLUDE "" CACHE PATH "KVAF device interface headers")
if(NOT EXISTS "${KVAF_INCLUDE}/kvaf_batch_get_device.h")
  message(FATAL_ERROR "Set KVAF_INCLUDE to the patched KVAF aiv_batch_get/include directory")
endif()
set(KVIP_FUSE_QUERY_WORK ON)
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set(KVIP_OUTPUT_WORKER ON)
option(KVIP_FINAL_TOPK_PRUNE "Filter final TopK candidates using the exact early threshold" OFF)
option(KVIP_SELECTED_METADATA "Read hot metadata only for final selected positions" OFF)
option(KVIP_EARLY_FETCH_READY "Publish safe Fetch descriptors before TND output completion" OFF)
option(KVIP_COMPACT_INIT "Use larger fill tiles and clear only live streaming change records" OFF)
option(KVIP_READY_BATCH_FETCH "Submit ready URMA rows without waiting for every row in the worker group" OFF)
option(KVIP_READY_DIRECT_RANGES "Use canonical descriptors for a single ready URMA range" OFF)
option(KVIP_ADAPTIVE_ADMISSION "Enable runtime URMA admission quotas and carry credits" OFF)
option(KVIP_LOW_MISS_SKIP "Skip speculative LI work when all runtime prefetch quotas are zero" OFF)
option(KVIP_LOW_MISS_TAIL "Clear only shadow keys and skip empty Index exchanges" OFF)
if(KVIP_LOW_MISS_TAIL AND NOT KVIP_LOW_MISS_SKIP)
  message(FATAL_ERROR "KVIP_LOW_MISS_TAIL requires KVIP_LOW_MISS_SKIP")
endif()
option(KVIP_AUTO_ROUTE "Choose the fixed-graph path using previous successful device feedback" OFF)
option(KVIP_LOW_MISS_SPARSE_INIT "Skip the unused history map reset on device-selected low rounds" OFF)
option(KVIP_AUTO_ROUTE_FAST_PROBE "Short-circuit read-only init probes once high routing is certain" OFF)
option(KVIP_AUTO_ROUTE_ROW_PUBLISH "Publish independent pending rows in parallel inside the existing init" OFF)
option(KVIP_AUTO_ROUTE_SIMT "Evaluate routing rows in parallel in the existing init" OFF)
option(KVIP_LOW_MISS_COMPACT_PLAN "Plan no-prefetch Q4 misses once in the existing resolver" OFF)
option(KVIP_LOW_MISS_DIRECT_INDEX "Use proven disjoint cold/hot Index swaps without permutation maps" OFF)
if(KVIP_AUTO_ROUTE_ROW_PUBLISH AND NOT KVIP_LOW_MISS_SPARSE_INIT)
  message(FATAL_ERROR "KVIP_AUTO_ROUTE_ROW_PUBLISH requires KVIP_LOW_MISS_SPARSE_INIT")
endif()
if(KVIP_AUTO_ROUTE_ROW_PUBLISH AND KVIP_AUTO_ROUTE_FAST_PROBE)
  message(FATAL_ERROR "Row publication requires complete evaluation, not fast probes")
endif()
if(KVIP_LOW_MISS_DIRECT_INDEX AND (NOT KVIP_LOW_MISS_TAIL OR NOT KVIP_SELECTED_METADATA OR NOT KVIP_EARLY_FETCH_READY OR NOT KVIP_OUTPUT_WORKER))
  message(FATAL_ERROR "KVIP_LOW_MISS_DIRECT_INDEX requires LOW_MISS_TAIL, SELECTED_METADATA, EARLY_FETCH_READY and OUTPUT_WORKER")
endif()
if(KVIP_AUTO_ROUTE_SIMT AND NOT KVIP_AUTO_ROUTE_ROW_PUBLISH)
  message(FATAL_ERROR "KVIP_AUTO_ROUTE_SIMT requires KVIP_AUTO_ROUTE_ROW_PUBLISH")
endif()
if(KVIP_LOW_MISS_COMPACT_PLAN AND (NOT KVIP_AUTO_ROUTE OR NOT KVIP_LOW_MISS_TAIL OR NOT KVIP_SELECTED_METADATA OR NOT KVIP_EARLY_FETCH_READY OR NOT KVIP_OUTPUT_WORKER))
  message(FATAL_ERROR "KVIP_LOW_MISS_COMPACT_PLAN requires AUTO_ROUTE, LOW_MISS_TAIL, SELECTED_METADATA, EARLY_FETCH_READY and OUTPUT_WORKER")
endif()
if(KVIP_AUTO_ROUTE_FAST_PROBE AND NOT KVIP_LOW_MISS_SPARSE_INIT)
  message(FATAL_ERROR "KVIP_AUTO_ROUTE_FAST_PROBE requires KVIP_LOW_MISS_SPARSE_INIT")
endif()
if(KVIP_LOW_MISS_SPARSE_INIT AND (NOT KVIP_AUTO_ROUTE OR NOT KVIP_LOW_MISS_TAIL OR NOT KVIP_SELECTED_METADATA))
  message(FATAL_ERROR "KVIP_LOW_MISS_SPARSE_INIT requires AUTO_ROUTE, LOW_MISS_TAIL and SELECTED_METADATA")
endif()
if(KVIP_AUTO_ROUTE AND (NOT KVIP_LOW_MISS_SKIP OR NOT KVIP_READY_BATCH_FETCH OR NOT KVIP_EARLY_FETCH_READY))
  message(FATAL_ERROR "KVIP_AUTO_ROUTE requires LOW_MISS_SKIP, READY_BATCH_FETCH and EARLY_FETCH_READY")
endif()
if(KVIP_LOW_MISS_SKIP AND NOT KVIP_ADAPTIVE_ADMISSION)
  message(FATAL_ERROR "KVIP_LOW_MISS_SKIP requires KVIP_ADAPTIVE_ADMISSION")
endif()
option(KVIP_PREFETCH_GATE "Enable a separate raised cold-prefetch score gate" OFF)
set(KVIP_PREFETCH_RANK 0 CACHE STRING "Early seed rank for speculative admission; 0 disables")
if(NOT KVIP_PREFETCH_RANK MATCHES "^(0|[1-9][0-9]*)$" OR KVIP_PREFETCH_RANK GREATER 2048)
  message(FATAL_ERROR "KVIP_PREFETCH_RANK must be 0 or 1..2048")
endif()
if(KVIP_PREFETCH_RANK AND NOT KVIP_PREFETCH_GATE)
  message(FATAL_ERROR "KVIP_PREFETCH_RANK requires KVIP_PREFETCH_GATE")
endif()
option(KVIP_SKIP_EXHAUSTED_PREFETCH "Skip candidate scanning after a permanent per-round limit is reached" OFF)
option(KVIP_INVERSE_MAPPING "Validate sparse prefetch mappings instead of clearing full history" OFF)
if(KVIP_LOW_MISS_SPARSE_INIT AND KVIP_INVERSE_MAPPING)
  message(FATAL_ERROR "KVIP_LOW_MISS_SPARSE_INIT and KVIP_INVERSE_MAPPING are separate initialization policies")
endif()
option(KVIP_EARLY_PREFETCH_DRAIN "Publish prefetch completion independently of final query planning" OFF)
option(KVIP_TOPK_PRUNE_DIAGNOSTICS "Independent final TopK pruning counters; never enable for headline timings" OFF)
if(KVIP_PREFETCH_GATE AND NOT KVIP_ADAPTIVE_ADMISSION)
  message(FATAL_ERROR "KVIP_PREFETCH_GATE requires KVIP_ADAPTIVE_ADMISSION")
endif()
if(KVIP_OUTPUT_WORKER AND NOT KVIP_FUSE_QUERY_WORK)
  message(FATAL_ERROR "KVIP_OUTPUT_WORKER requires KVIP_FUSE_QUERY_WORK")
endif()
set(_base "${CMAKE_CURRENT_SOURCE_DIR}")
ascendc_library(v9_kernel STATIC op_kernel/indexer.cpp op_kernel/lifecycle.cpp op_kernel/streaming.cpp)
ascendc_include_directories(v9_kernel PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/op_kernel
  ${CMAKE_CURRENT_SOURCE_DIR}/op_kernel/indexer ${KVAF_INCLUDE}
  ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(v9_kernel PRIVATE -DKVIP_STREAMING_LI=1 -DKVIP_OUTPUT_WORKER=$<BOOL:${KVIP_OUTPUT_WORKER}> -DKVIP_FUSE_QUERY_WORK=$<BOOL:${KVIP_FUSE_QUERY_WORK}> -O2 -g --cce-auto-sync=off
  -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232
  -DV9_PARTIAL_SCAN_SIMD=1 -DV9_PROBE_EPOCH=0 -DV9_INTERMEDIATE_PROFILING=0
  -DV9_BATCH_POST=2 -DSIM_Q=4 -DSIM_CAPTURE_HITS=1 -DV9_SIMULATION=0 -DROTATION_PUBLISH_TILES=1)
foreach(option FINAL_TOPK_PRUNE SELECTED_METADATA EARLY_FETCH_READY COMPACT_INIT READY_BATCH_FETCH READY_DIRECT_RANGES ADAPTIVE_ADMISSION LOW_MISS_SKIP LOW_MISS_TAIL LOW_MISS_SPARSE_INIT AUTO_ROUTE_FAST_PROBE AUTO_ROUTE_ROW_PUBLISH AUTO_ROUTE_SIMT LOW_MISS_DIRECT_INDEX LOW_MISS_COMPACT_PLAN AUTO_ROUTE PREFETCH_GATE SKIP_EXHAUSTED_PREFETCH EARLY_PREFETCH_DRAIN INVERSE_MAPPING TOPK_PRUNE_DIAGNOSTICS)
  ascendc_compile_options(v9_kernel PRIVATE -DKVIP_${option}=$<BOOL:${KVIP_${option}}>)
endforeach()
ascendc_compile_options(v9_kernel PRIVATE -DKVIP_PREFETCH_RANK=${KVIP_PREFETCH_RANK})
add_library(v9_bridge SHARED op_host/launch.cpp)
target_include_directories(v9_bridge PRIVATE ${KVAF_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/include ${_base}/include)
target_compile_definitions(v9_bridge PRIVATE KVIP_OUTPUT_WORKER=$<BOOL:${KVIP_OUTPUT_WORKER}>)
foreach(option FINAL_TOPK_PRUNE SELECTED_METADATA EARLY_FETCH_READY COMPACT_INIT READY_BATCH_FETCH READY_DIRECT_RANGES ADAPTIVE_ADMISSION LOW_MISS_SKIP LOW_MISS_TAIL LOW_MISS_SPARSE_INIT AUTO_ROUTE_FAST_PROBE AUTO_ROUTE_ROW_PUBLISH AUTO_ROUTE_SIMT LOW_MISS_DIRECT_INDEX LOW_MISS_COMPACT_PLAN AUTO_ROUTE PREFETCH_GATE SKIP_EXHAUSTED_PREFETCH EARLY_PREFETCH_DRAIN INVERSE_MAPPING TOPK_PRUNE_DIAGNOSTICS)
  target_compile_definitions(v9_bridge PRIVATE KVIP_${option}=$<BOOL:${KVIP_${option}}>)
endforeach()
target_compile_definitions(v9_bridge PRIVATE KVIP_PREFETCH_RANK=${KVIP_PREFETCH_RANK})
target_link_libraries(v9_bridge PRIVATE v9_kernel host_intf_pub ascendcl)
