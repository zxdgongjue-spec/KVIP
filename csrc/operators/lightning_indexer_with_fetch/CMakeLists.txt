cmake_minimum_required(VERSION 3.16)
project(v9_rotating_fetch LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUN_MODE npu CACHE STRING "npu or sim")
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "A5 SoC")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann CACHE PATH "CANN installation")
set(INDEX_HEADS 64 CACHE STRING "Indexer query heads: 32 or 64")
if(NOT INDEX_HEADS MATCHES "^(32|64)$")
  message(FATAL_ERROR "INDEX_HEADS must be 32 or 64")
endif()
set(SIM_BATCH 2 CACHE STRING "Batch, 1..32")
set(SIM_HISTORY 131072 CACHE STRING "History tokens")
set(SIM_RESIDENT 12288 CACHE STRING "Hot prefix tokens")
set(TOKEN_BYTES 656 CACHE STRING "Target KV bytes per token: 656 or 2624")
if(NOT TOKEN_BYTES MATCHES "^(656|2624)$")
  message(FATAL_ERROR "TOKEN_BYTES must be 656 or 2624")
endif()
set(LI_TILING auto CACHE STRING "auto: native Q2/Q4 split; q4: force prior one-AIC/batch mode")
if(NOT LI_TILING MATCHES "^(auto|q4)$")
  message(FATAL_ERROR "LI_TILING must be auto or q4")
endif()
set(AIC_COUNT "" CACHE STRING "Available AICs; required for NPU builds, simulator defaults to 32")
if(AIC_COUNT STREQUAL "")
  if(RUN_MODE STREQUAL "sim")
    set(AIC_COUNT 32)
  else()
    message(FATAL_ERROR "Set AIC_COUNT from aclrtGetDeviceInfo(ACL_DEV_ATTR_CUBE_CORE_NUM)")
  endif()
endif()
if(NOT AIC_COUNT MATCHES "^[0-9]+$" OR AIC_COUNT LESS 1 OR AIC_COUNT GREATER 32)
  message(FATAL_ERROR "AIC_COUNT must be 1..32")
endif()
math(EXPR SPLIT_TASKS "${SIM_BATCH} * 2")
set(V9_SPLIT_Q 0)
set(FORCE_Q4 1)
if(LI_TILING STREQUAL "auto" AND SPLIT_TASKS LESS_EQUAL AIC_COUNT)
  set(V9_SPLIT_Q 1)
  set(FORCE_Q4 0)
endif()
set(INTERMEDIATE_FETCHES 32 CACHE STRING "Requested intermediate windows; clipped to actual LI AIC count, 0 disables")
if(NOT INTERMEDIATE_FETCHES MATCHES "^[0-9]+$")
  message(FATAL_ERROR "INTERMEDIATE_FETCHES must be a nonnegative integer")
endif()
math(EXPR LI_CORES "${SIM_BATCH} * (1 + ${V9_SPLIT_Q})")
if(LI_CORES GREATER AIC_COUNT)
  set(LI_CORES ${AIC_COUNT})
endif()
set(SERVICE_ROUNDS ${INTERMEDIATE_FETCHES})
if(SERVICE_ROUNDS GREATER LI_CORES)
  set(SERVICE_ROUNDS ${LI_CORES})
endif()
set(PARTIAL_SCAN_SIMD 1 CACHE STRING "SIMD predicate clipping for unequal query bounds (0/1)")
if(NOT PARTIAL_SCAN_SIMD MATCHES "^[01]$")
  message(FATAL_ERROR "PARTIAL_SCAN_SIMD must be 0/1")
endif()
set(PROBE_EPOCH 0 CACHE STRING "Standalone probe window, simulation only")
if(NOT PROBE_EPOCH MATCHES "^[0-9]+$" OR (PROBE_EPOCH GREATER 0 AND (NOT RUN_MODE STREQUAL "sim" OR PROBE_EPOCH GREATER_EQUAL SERVICE_ROUNDS)))
  message(FATAL_ERROR "PROBE_EPOCH requires an existing simulation service window")
endif()
set(INTERMEDIATE_PROFILING 1 CACHE STRING "Compile intermediate Fetch timing and pure counters (0/1)")
if(NOT INTERMEDIATE_PROFILING MATCHES "^[01]$")
  message(FATAL_ERROR "INTERMEDIATE_PROFILING must be 0/1")
endif()
set(FINAL_REPLICATED 1 CACHE STRING "Replicate final planning (1) or partition and synchronize (0)")
if(NOT FINAL_REPLICATED MATCHES "^[01]$")
  message(FATAL_ERROR "FINAL_REPLICATED must be 0/1")
endif()
set(FINAL_AIVS 0 CACHE STRING "AIVs per batch for final resolve; 0 auto, 1/2/4 explicit")
if(NOT FINAL_AIVS MATCHES "^(0|1|2|4)$")
  message(FATAL_ERROR "FINAL_AIVS must be 0/1/2/4")
endif()
set(SERVICE_AICS 0 CACHE STRING "AICs per intermediate window; 0 uses active AICs/windows")
if(NOT SERVICE_AICS MATCHES "^[0-9]+$")
  message(FATAL_ERROR "SERVICE_AICS must be nonnegative")
endif()
math(EXPR SERVICE_TOTAL "${SERVICE_AICS} * ${SERVICE_ROUNDS}")
if(SERVICE_TOTAL GREATER LI_CORES)
  message(FATAL_ERROR "SERVICE_AICS exceeds active AIC budget")
endif()
set(FETCH_POST_MODE batch_mixed CACHE STRING "batch_simt, batch_mixed or per_batch_simd")
if(FETCH_POST_MODE STREQUAL "batch_simt")
  set(V9_BATCH_POST 1)
elseif(FETCH_POST_MODE STREQUAL "batch_mixed")
  set(V9_BATCH_POST 2)
elseif(FETCH_POST_MODE STREQUAL "per_batch_simd")
  set(V9_BATCH_POST 0)
else()
  message(FATAL_ERROR "FETCH_POST_MODE must be batch_simt, batch_mixed or per_batch_simd")
endif()
set(ROTATION_PUBLISH_TILES 1 CACHE STRING "Ready publication interval")
set(KVAF_INCLUDE "" CACHE PATH "KVAF batch_get_abi.h and kvaf_batch_get_device.h directory")
if(NOT EXISTS "${KVAF_INCLUDE}/kvaf_batch_get_device.h")
  message(FATAL_ERROR "Set KVAF_INCLUDE to the real KVAF aiv_batch_get/include directory")
endif()
math(EXPR HR "${SIM_HISTORY} % 256")
math(EXPR RR "${SIM_RESIDENT} % 256")
math(EXPR COLD_TILES "(${SIM_HISTORY} - ${SIM_RESIDENT}) / 128")
if(SIM_BATCH LESS 1 OR SIM_BATCH GREATER 32 OR NOT HR EQUAL 0 OR NOT RR EQUAL 0 OR SIM_RESIDENT LESS 2048 OR COLD_TILES LESS_EQUAL SERVICE_ROUNDS OR COLD_TILES LESS 1)
  message(FATAL_ERROR "Requires B1..32 Q4, 256-aligned H>R>=2048 and enough cold tiles for distinct windows")
endif()
math(EXPR FINAL_UB "196608 + ((${SIM_HISTORY} + 31) / 32) * 4 + 256")
if(FINAL_UB GREATER 223232)
  message(FATAL_ERROR "History exceeds bounded final resolver UB; required ${FINAL_UB}")
endif()
if(NOT ROTATION_PUBLISH_TILES MATCHES "^(1|2|4|8|16)$")
  message(FATAL_ERROR "Ready group must be 1,2,4,8,16 tiles")
endif()
if(RUN_MODE STREQUAL "sim")
 set(V9_SIMULATION 1)
else()
 set(V9_SIMULATION 0)
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(v9_kernel STATIC indexer.cpp lifecycle.cpp transport.cpp)
ascendc_include_directories(v9_kernel PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${KVAF_INCLUDE}
  ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
# Keep the compiler UB reservation and source allocation bound identical.
# CANN 9.1 adds 6 KiB: 218 KiB user + 6 KiB compiler = 224 KiB SIMT limit.
set(RESOLVER_COMPARE 0 CACHE STRING "Build resolver-only comparison entries")
set(V9_USER_UB_BYTES 223232)
ascendc_compile_options(v9_kernel PRIVATE -DV9_RESOLVER_COMPARE=${RESOLVER_COMPARE} -O2 -g --cce-auto-sync=off
  -mllvm -user-reserved-ub-size=${V9_USER_UB_BYTES} -DV9_USER_UB_BYTES=${V9_USER_UB_BYTES} -DV9_TOKEN_BYTES=${TOKEN_BYTES} -DINDEX_HEADS=${INDEX_HEADS} -DSIM_BATCH=${SIM_BATCH} -DSIM_HISTORY=${SIM_HISTORY}
  -DV9_PARTIAL_SCAN_SIMD=${PARTIAL_SCAN_SIMD} -DV9_PROBE_EPOCH=${PROBE_EPOCH} -DV9_INTERMEDIATE_PROFILING=${INTERMEDIATE_PROFILING} -DV9_FINAL_REPLICATED=${FINAL_REPLICATED} -DV9_FINAL_AIVS=${FINAL_AIVS} -DV9_SERVICE_AICS=${SERVICE_AICS} -DV9_BATCH_POST=${V9_BATCH_POST} -DSIM_RESIDENT=${SIM_RESIDENT} -DSIM_Q=4 -DSIM_CAPTURE_HITS=1 -DV9_SIMULATION=${V9_SIMULATION}
  -DROTATION_PUBLISH_TILES=${ROTATION_PUBLISH_TILES} -DV9_AIC_COUNT=${AIC_COUNT} -DV9_SPLIT_Q=${V9_SPLIT_Q} -DV9_INTERMEDIATE_FETCHES=${SERVICE_ROUNDS}
 )
add_library(v9_bridge SHARED launch.cpp)
target_include_directories(v9_bridge PRIVATE ${KVAF_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(v9_bridge PRIVATE V9_RESOLVER_COMPARE=${RESOLVER_COMPARE} V9_TOKEN_BYTES=${TOKEN_BYTES} V9_FINAL_REPLICATED=${FINAL_REPLICATED} V9_FINAL_AIVS=${FINAL_AIVS} V9_SERVICE_AICS=${SERVICE_AICS} V9_AIC_COUNT=${AIC_COUNT} V9_INTERMEDIATE_FETCHES=${SERVICE_ROUNDS} V9_SPLIT_Q=${V9_SPLIT_Q} V9_SIMULATION=${V9_SIMULATION} SIM_BATCH=${SIM_BATCH} SIM_HISTORY=${SIM_HISTORY} SIM_RESIDENT=${SIM_RESIDENT})
target_link_libraries(v9_bridge PRIVATE v9_kernel host_intf_pub ascendcl)

# Independent AIV target: does not change the LI/Fetch compilation options.
ascendc_library(kvop_cache_update_kernel STATIC index_cache_update.cpp)
ascendc_include_directories(kvop_cache_update_kernel PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(kvop_cache_update_kernel PRIVATE -O2
  -mllvm -user-reserved-ub-size=223232 -DSIM_HISTORY=${SIM_HISTORY} -DSIM_RESIDENT=${SIM_RESIDENT})
add_library(kvop_cache_update SHARED index_cache_update_host.cpp)
target_compile_definitions(kvop_cache_update PRIVATE SIM_HISTORY=${SIM_HISTORY} SIM_RESIDENT=${SIM_RESIDENT})
target_link_libraries(kvop_cache_update PRIVATE kvop_cache_update_kernel host_intf_pub ascendcl)

target_link_libraries(v9_bridge PRIVATE kvop_cache_update)
