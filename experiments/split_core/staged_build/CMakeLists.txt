cmake_minimum_required(VERSION 3.16)
project(kvip_staged_routes LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim CACHE STRING "sim or npu")
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "Target SoC")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann-9.1.0 CACHE PATH "CANN")
set(STAGED_SOURCE "" CACHE PATH "prepare_staged_source.py output")
set(SCORE_VERSION current CACHE STRING "score producer version")
set(KVAF_INCLUDE "" CACHE PATH "patched KVAF headers")
if(NOT EXISTS "${STAGED_SOURCE}/staged-source.json")
 message(FATAL_ERROR "Prepare isolated staged source")
endif()
if(NOT RUN_MODE STREQUAL sim)
 message(FATAL_ERROR "This functional driver requires Model; native transport uses the production bridge")
endif()
file(READ "${STAGED_SOURCE}/staged-source.json" _source_manifest)
string(JSON _producer GET "${_source_manifest}" producer)
string(JSON _transport GET "${_source_manifest}" model_transport)
if(NOT _producer STREQUAL SCORE_VERSION OR NOT _transport STREQUAL "synchronous GM adapter")
 message(FATAL_ERROR "Driver version/transport differs from prepared source")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set(PRODUCER ${STAGED_SOURCE}/producer/csrc/operators/lightning_indexer_with_fetch)
set(TAIL ${STAGED_SOURCE}/tail/csrc/operators/lightning_indexer_with_fetch)
ascendc_library(producer STATIC ../score_kernel.cpp)
ascendc_library(tail STATIC ${TAIL}/op_kernel/indexer.cpp ${TAIL}/op_kernel/staged_vector.cpp
 ${TAIL}/op_kernel/lifecycle.cpp ${TAIL}/op_kernel/streaming.cpp)
foreach(target producer tail)
 if(target STREQUAL producer)
  set(OP ${PRODUCER})
  if(SCORE_VERSION STREQUAL historical)
   set(STREAMING 0)
  else()
   set(STREAMING 1)
  endif()
 else()
  set(OP ${TAIL})
  set(STREAMING 1)
 endif()
 ascendc_include_directories(${target} PRIVATE ${OP}/include ${OP}/op_kernel ${OP}/op_kernel/indexer
  ${KVAF_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/../historical/full ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
 ascendc_compile_options(${target} PRIVATE -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232
  -DV9_USER_UB_BYTES=223232 -DKVIP_STREAMING_LI=${STREAMING} -DSIM_Q=4 -DSIM_CAPTURE_HITS=1
  -DKVIP_OUTPUT_WORKER=1 -DKVIP_FUSE_QUERY_WORK=1 -DV9_SIMULATION=0 -DROTATION_PUBLISH_TILES=1
  -DKVIP_FUSION3=0 -DKVIP_FUSION3_TEST_TRANSPORT=1 -DKVIP_OPT_LI_REGISTER_PREDICATE=0
  -DV9_PARTIAL_SCAN_SIMD=1 -DV9_PROBE_EPOCH=0 -DV9_INTERMEDIATE_PROFILING=0 -DV9_BATCH_POST=2)
endforeach()
add_executable(staged_routes ../staged_host.cpp ${TAIL}/op_host/launch.cpp)
target_include_directories(staged_routes PRIVATE ${TAIL}/include ${TAIL}/op_kernel ${KVAF_INCLUDE}
 ${CMAKE_CURRENT_SOURCE_DIR}/../historical/full ${STAGED_SOURCE}/tail/csrc/operators/common/lifetch)
# Same integration function, selecting the independently compiled versioned producer.
target_compile_definitions(staged_routes PRIVATE KVIP_OUTPUT_WORKER=1
 aclrtlaunch_v9_staged_score=aclrtlaunch_split_score_only SCORE_VERSION="${SCORE_VERSION}" STAGED_RUN_MODE="${RUN_MODE}")
target_link_libraries(staged_routes PRIVATE producer tail host_intf_pub ascendcl)
if(RUN_MODE STREQUAL sim)
 target_link_options(staged_routes PRIVATE -Wl,--allow-shlib-undefined)
endif()
