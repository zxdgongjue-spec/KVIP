cmake_minimum_required(VERSION 3.16)
project(kvip_historical_split LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim CACHE STRING "CA Model only")
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "Actual compile target")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann CACHE PATH "Matching CANN")
set(WORK_ROOT "" CACHE PATH "prepare_historical.py output")
set(KVAF_INCLUDE "" CACHE PATH "Prepared pinned KVAF aiv_batch_get/include")
if(NOT EXISTS "${WORK_ROOT}/prepared.json" OR NOT EXISTS "${KVAF_INCLUDE}/kvaf_batch_get_device.h")
  message(FATAL_ERROR "Prepare the historical source/fixture and patched KVAF headers first")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set(HISTORICAL_VARIANT history CACHE STRING "full or history; use a separate build directory per variant")
if(NOT HISTORICAL_VARIANT STREQUAL "full" AND NOT HISTORICAL_VARIANT STREQUAL "history")
  message(FATAL_ERROR "HISTORICAL_VARIANT must be full or history")
endif()
set(OP csrc/operators/lightning_indexer_with_fetch)
set(BASE ${WORK_ROOT}/full-source/${OP})
set(OPTIONS -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232
  -DV9_USER_UB_BYTES=223232 -DKVIP_OPT_LI_REGISTER_PREDICATE=0
  -DKVIP_FUSION3=0 -DKVIP_FUSION3_TEST_TRANSPORT=1 -DV9_RESOLVER_COMPARE=0
  -DV9_PARTIAL_SCAN_SIMD=1 -DV9_PROBE_EPOCH=0 -DV9_INTERMEDIATE_PROFILING=0
  -DV9_BATCH_POST=2 -DSIM_Q=4 -DV9_SIMULATION=0 -DROTATION_PUBLISH_TILES=1)
foreach(VARIANT IN ITEMS full pooled history)
  if((HISTORICAL_VARIANT STREQUAL "full" AND VARIANT STREQUAL "history") OR
     (HISTORICAL_VARIANT STREQUAL "history" AND VARIANT STREQUAL "pooled"))
    continue()
  endif()
  if(VARIANT STREQUAL "full")
    set(SOURCE ${BASE})
    ascendc_library(full_kernel STATIC ${SOURCE}/op_kernel/indexer.cpp ${SOURCE}/op_kernel/lifecycle.cpp)
  else()
    set(SOURCE ${WORK_ROOT}/${VARIANT}-source/${OP})
    ascendc_library(${VARIANT}_kernel STATIC ${SOURCE}/op_kernel/indexer.cpp)
  endif()
  ascendc_include_directories(${VARIANT}_kernel PRIVATE ${SOURCE}/include ${SOURCE}/op_kernel
    ${SOURCE}/op_kernel/indexer ${KVAF_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/full
    ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
  if(VARIANT STREQUAL "history")
    ascendc_compile_options(${VARIANT}_kernel PRIVATE ${OPTIONS} -DSIM_CAPTURE_HITS=0)
  else()
    ascendc_compile_options(${VARIANT}_kernel PRIVATE ${OPTIONS} -DSIM_CAPTURE_HITS=1)
  endif()
endforeach()
ascendc_library(update_kernel STATIC ${BASE}/op_kernel/index_cache_update.cpp)
ascendc_include_directories(update_kernel PRIVATE ${BASE}/include ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(update_kernel PRIVATE -O2 -g -mllvm -user-reserved-ub-size=223232)
foreach(VARIANT IN ITEMS full history)
  if(NOT VARIANT STREQUAL HISTORICAL_VARIANT)
    continue()
  endif()
  if(VARIANT STREQUAL "full")
    set(TOPK_SOURCE topk_v2.cpp)
    set(HOST full/full_pool_host.cpp)
    set(PRODUCER pooled_kernel)
    set(EXECUTABLE full_pool_sim)
  else()
    set(TOPK_SOURCE history/history_topk_kernel.cpp)
    set(HOST history/history_host.cpp)
    set(PRODUCER history_kernel)
    set(EXECUTABLE history_pool_sim)
  endif()
  ascendc_library(${VARIANT}_topk STATIC ${TOPK_SOURCE})
  ascendc_include_directories(${VARIANT}_topk PRIVATE ${BASE}/op_kernel/indexer ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
  ascendc_compile_options(${VARIANT}_topk PRIVATE -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232)
  add_executable(${EXECUTABLE} ${HOST} ${BASE}/op_host/launch.cpp ${BASE}/op_host/index_cache_update.cpp)
  target_include_directories(${EXECUTABLE} PRIVATE ${BASE}/include ${KVAF_INCLUDE})
  target_compile_definitions(${EXECUTABLE} PRIVATE KVIP_FUSION3=0 V9_RESOLVER_COMPARE=0)
  target_link_libraries(${EXECUTABLE} PRIVATE ${PRODUCER} ${VARIANT}_topk full_kernel update_kernel host_intf_pub ascendcl)
  target_link_options(${EXECUTABLE} PRIVATE -Wl,--allow-shlib-undefined)
endforeach()
