cmake_minimum_required(VERSION 3.19)
project(li_qk_controls LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE npu CACHE STRING "npu or sim")
set(SOC_VERSION "" CACHE STRING "Actual A5 SoC")
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "Matching CANN installation")
set(SCORE_SOURCE "" CACHE PATH "Prepared source snapshot")
set(KVAF_INCLUDE "" CACHE PATH "Existing compatible device headers; no Fetch is launched")
if(NOT EXISTS "${SCORE_SOURCE}/score-source.json" OR NOT EXISTS "${KVAF_INCLUDE}/kvaf_batch_get_device.h")
 message(FATAL_ERROR "Prepare source and provide existing KVAF include directory")
endif()
if(NOT SOC_VERSION OR NOT EXISTS "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
 message(FATAL_ERROR "Specify actual SOC_VERSION and matching CANN path")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set(OP ${SCORE_SOURCE}/csrc/operators/lightning_indexer_with_fetch)
ascendc_library(li_score STATIC score.cpp)
ascendc_library(li_post STATIC post.cpp)
option(LI_TAIL_CONTROLS "Build isolated post-score controls" OFF)
set(LI_DEVICE_TARGETS li_score li_post)
option(LI_PRODUCER_CONTROLS "Build fused producer controls" OFF)
if(LI_PRODUCER_CONTROLS)
 ascendc_library(li_producer_kernel STATIC producer.cpp)
 ascendc_library(li_producer_tail STATIC producer_tail.cpp)
 list(APPEND LI_DEVICE_TARGETS li_producer_kernel li_producer_tail)
 add_executable(li_producer producer_host.cpp)
 target_include_directories(li_producer PRIVATE ${OP}/include ${OP}/op_kernel ${KVAF_INCLUDE} ${SCORE_SOURCE}/csrc/operators/common/lifetch ${CMAKE_CURRENT_SOURCE_DIR}/../split_core/model_support)
 target_compile_definitions(li_producer PRIVATE LI_QK_RUN_MODE="${RUN_MODE}")
 target_link_libraries(li_producer PRIVATE li_producer_kernel li_producer_tail li_score li_post host_intf_pub ascendcl)
 if(RUN_MODE STREQUAL sim)
  target_link_options(li_producer PRIVATE -Wl,--allow-shlib-undefined)
 endif()
endif()
option(LI_RADIX_CONTROLS "Build exact distributed radix controls" OFF)
if(LI_RADIX_CONTROLS)
 ascendc_library(li_radix_kernel STATIC radix.cpp)
 list(APPEND LI_DEVICE_TARGETS li_radix_kernel)
endif()
if(LI_TAIL_CONTROLS)
 ascendc_library(li_tail_kernel STATIC tail.cpp)
 list(APPEND LI_DEVICE_TARGETS li_tail_kernel)
endif()
if(LI_RADIX_CONTROLS)
 add_executable(li_qk_radix host.cpp)
 target_include_directories(li_qk_radix PRIVATE ${OP}/include ${OP}/op_kernel ${KVAF_INCLUDE}
  ${SCORE_SOURCE}/csrc/operators/common/lifetch ${CMAKE_CURRENT_SOURCE_DIR}/../split_core/model_support)
 target_compile_definitions(li_qk_radix PRIVATE LI_QK_RUN_MODE="${RUN_MODE}" LI_QK_RADIX_COMPARE=1)
 target_link_libraries(li_qk_radix PRIVATE li_score li_post li_radix_kernel host_intf_pub ascendcl)
 if(RUN_MODE STREQUAL sim)
  target_link_options(li_qk_radix PRIVATE -Wl,--allow-shlib-undefined)
 endif()
 add_executable(li_radix radix_host.cpp)
 target_include_directories(li_radix PRIVATE ${OP}/include ${OP}/op_kernel ${KVAF_INCLUDE}
  ${SCORE_SOURCE}/csrc/operators/common/lifetch ${CMAKE_CURRENT_SOURCE_DIR}/../split_core/model_support)
 target_compile_definitions(li_radix PRIVATE LI_QK_RUN_MODE="${RUN_MODE}")
 target_link_libraries(li_radix PRIVATE li_radix_kernel li_post host_intf_pub ascendcl)
 if(RUN_MODE STREQUAL sim)
  target_link_options(li_radix PRIVATE -Wl,--allow-shlib-undefined)
 endif()
endif()
foreach(target ${LI_DEVICE_TARGETS})
 ascendc_include_directories(${target} PRIVATE ${OP}/include ${OP}/op_kernel ${OP}/op_kernel/indexer
  ${KVAF_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../split_core/model_support
  ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
 ascendc_compile_options(${target} PRIVATE -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232
  -DV9_USER_UB_BYTES=223232 -DKVIP_STREAMING_LI=0 -DSIM_Q=4 -DSIM_CAPTURE_HITS=0
  -DKVIP_OUTPUT_WORKER=1 -DKVIP_FUSE_QUERY_WORK=1 -DV9_SIMULATION=0 -DROTATION_PUBLISH_TILES=1
  -DKVIP_FUSION3=0 -DKVIP_FUSION3_TEST_TRANSPORT=1 -DKVIP_OPT_LI_REGISTER_PREDICATE=0
  -DV9_PARTIAL_SCAN_SIMD=1 -DV9_PROBE_EPOCH=0 -DV9_INTERMEDIATE_PROFILING=0 -DV9_BATCH_POST=2)
endforeach()
add_executable(li_qk host.cpp)
target_include_directories(li_qk PRIVATE ${OP}/include ${OP}/op_kernel ${KVAF_INCLUDE}
 ${SCORE_SOURCE}/csrc/operators/common/lifetch ${CMAKE_CURRENT_SOURCE_DIR}/../split_core/model_support)
target_compile_definitions(li_qk PRIVATE LI_QK_RUN_MODE="${RUN_MODE}")
target_link_libraries(li_qk PRIVATE li_score li_post host_intf_pub ascendcl)
if(RUN_MODE STREQUAL sim)
 target_link_options(li_qk PRIVATE -Wl,--allow-shlib-undefined)
endif()
if(LI_TAIL_CONTROLS)
 add_executable(li_tail tail_host.cpp)
 target_include_directories(li_tail PRIVATE ${OP}/include ${OP}/op_kernel ${KVAF_INCLUDE}
  ${SCORE_SOURCE}/csrc/operators/common/lifetch ${CMAKE_CURRENT_SOURCE_DIR}/../split_core/model_support)
 target_compile_definitions(li_tail PRIVATE LI_QK_RUN_MODE="${RUN_MODE}")
 target_link_libraries(li_tail PRIVATE li_tail_kernel host_intf_pub ascendcl)
 if(RUN_MODE STREQUAL sim)
  target_link_options(li_tail PRIVATE -Wl,--allow-shlib-undefined)
 endif()
endif()
