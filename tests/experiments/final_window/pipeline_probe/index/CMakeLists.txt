cmake_minimum_required(VERSION 3.16)
project(real_topk LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim)
set(SOC_VERSION Ascend950PR_9599)
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann)
set(OP_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/csrc/operators/lightning_indexer_with_fetch)
set(KVAF_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/kvaf-include)
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(real_topk_kernel STATIC kernel.cpp)
ascendc_include_directories(real_topk_kernel PRIVATE ${OP_ROOT}/include ${OP_ROOT}/op_kernel ${OP_ROOT}/op_kernel/indexer ${KVAF_INCLUDE} ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(real_topk_kernel PRIVATE -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232 -DV9_BATCH_POST=2 -DSIM_Q=4 -DSIM_CAPTURE_HITS=1 -DV9_SIMULATION=1 -DKVIP_STREAMING_LI=1 -DKVIP_OUTPUT_WORKER=1 -DROTATION_PUBLISH_TILES=1)
add_executable(real_topk_sim driver.cpp)
target_include_directories(real_topk_sim PRIVATE ${OP_ROOT}/include ${OP_ROOT}/op_kernel ${KVAF_INCLUDE})
target_link_libraries(real_topk_sim PRIVATE real_topk_kernel host_intf_pub ascendcl)
target_link_options(real_topk_sim PRIVATE -Wl,--allow-shlib-undefined)
target_compile_definitions(real_topk_sim PRIVATE KVIP_OUTPUT_WORKER=1)
