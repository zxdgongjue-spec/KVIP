cmake_minimum_required(VERSION 3.16)
project(li_candidates LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim CACHE STRING "sim or npu")
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "SoC")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann CACHE PATH "CANN")
set(KVAF_INCLUDE "" CACHE PATH "KVAF device headers")
if(NOT EXISTS "${KVAF_INCLUDE}/batch_get_abi.h")
  message(FATAL_ERROR "Set KVAF_INCLUDE to the patched KVAF aiv_batch_get/include directory")
endif()
set(COMPACT_OUTPUT 0 CACHE STRING "0: native bitmap; 1: packed per-tile candidates")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare.py ${CMAKE_CURRENT_BINARY_DIR}/source RESULT_VARIABLE prepare_result)
if(NOT prepare_result EQUAL 0)
  message(FATAL_ERROR "Failed to prepare isolated LI sources")
endif()
set(OP ${CMAKE_CURRENT_BINARY_DIR}/source/csrc/operators/lightning_indexer_with_fetch)
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(li_kernel STATIC kernel.cpp)
ascendc_include_directories(li_kernel PRIVATE ${OP}/include ${OP}/op_kernel ${OP}/op_kernel/indexer ${KVAF_INCLUDE} ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(li_kernel PRIVATE -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232 -DV9_BATCH_POST=2 -DSIM_Q=4 -DSIM_CAPTURE_HITS=1 -DV9_SIMULATION=0 -DROTATION_PUBLISH_TILES=1 -DEXPERIMENT_COMPACT=${COMPACT_OUTPUT})
add_executable(li_candidate_sim main.cpp)
target_include_directories(li_candidate_sim PRIVATE ${OP}/include ${KVAF_INCLUDE})
target_compile_definitions(li_candidate_sim PRIVATE EXPERIMENT_COMPACT=${COMPACT_OUTPUT})
target_link_libraries(li_candidate_sim PRIVATE li_kernel host_intf_pub ascendcl)
target_link_options(li_candidate_sim PRIVATE -Wl,--allow-shlib-undefined)
