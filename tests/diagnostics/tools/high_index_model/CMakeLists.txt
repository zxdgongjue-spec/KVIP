cmake_minimum_required(VERSION 3.16)
project(high_index_case LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "")
set(RUN_MODE npu CACHE STRING "")
set(SOC_VERSION Ascend950DT_9582 CACHE STRING "")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann-9.2.0 CACHE PATH "")
set(OP "${CMAKE_CURRENT_SOURCE_DIR}/csrc/operators/lightning_indexer_with_fetch")
if(NOT IS_DIRECTORY "${KVAF_INCLUDE}")
  message(FATAL_ERROR "KVAF_INCLUDE must name the patched KVAF AIV include directory")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(tail_kernel STATIC ${OP}/op_kernel/streaming.cpp)
ExternalProject_Add_Step(tail_kernel_preprocess schedule
 COMMAND ${CMAKE_COMMAND} "-DGENERATED_SOURCE=${CMAKE_BINARY_DIR}/auto_gen/tail_kernel/auto_gen_streaming.cpp" -P "${OP}/cmake/set_tail_schedule.cmake"
 DEPENDEES build DEPENDERS install ALWAYS TRUE)
ascendc_include_directories(tail_kernel PRIVATE ${OP}/include ${OP}/op_kernel ${OP}/op_kernel/indexer ${KVAF_INCLUDE})
ascendc_compile_options(tail_kernel PRIVATE -O2 -g --cce-auto-sync=off --cce-res-usage -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232
 -DKVIP_STREAMING_LI=1 -DKVIP_OUTPUT_WORKER=1 -DKVIP_FUSE_QUERY_WORK=1 -DV9_PARTIAL_SCAN_SIMD=1 -DV9_PROBE_EPOCH=0 -DV9_INTERMEDIATE_PROFILING=0 -DV9_BATCH_POST=2 -DSIM_Q=4 -DSIM_CAPTURE_HITS=1 -DV9_SIMULATION=0 -DROTATION_PUBLISH_TILES=1 -DKVIP_FINAL_TOPK_PRUNE=0 -DKVIP_SELECTED_METADATA=1 -DKVIP_EARLY_FETCH_READY=1 -DKVIP_COMPACT_INIT=0 -DKVIP_READY_BATCH_FETCH=1 -DKVIP_LOW_MISS_SKIP=1 -DKVIP_AUTO_ROUTE=1 -DKVIP_LOW_MISS_TAIL=1 -DKVIP_LOW_MISS_SPARSE_INIT=1 -DKVIP_AUTO_ROUTE_ROW_PUBLISH=1 -DKVIP_LOW_MISS_DIRECT_INDEX=1 -DKVIP_LOW_MISS_POST_INDEX=1 -DKVIP_ADAPTIVE_ADMISSION=1)
if(RUN_MODE STREQUAL "sim")
  target_link_directories(tail_kernel PUBLIC ${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib)
endif()
add_executable(high_index_case main.cpp)
target_include_directories(high_index_case PRIVATE ${OP}/include ${OP}/op_kernel ${KVAF_INCLUDE})
target_compile_definitions(high_index_case PRIVATE KVIP_OUTPUT_WORKER=1)
if(EXISTS "${OP}/op_kernel/high_index_cold_plan.h")
  target_compile_definitions(high_index_case PRIVATE INDEX_COLD_PLAN=1)
endif()
if(RUN_MODE STREQUAL "npu")
  target_compile_definitions(high_index_case PRIVATE INDEX_TARGET_NPU=1)
endif()
target_link_libraries(high_index_case PRIVATE tail_kernel)
target_link_options(high_index_case PRIVATE "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/lib64" "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib")
