cmake_minimum_required(VERSION 3.16)
project(kvip_pipeline_ab LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim CACHE STRING "Execution target")
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "Target SoC")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann-9.1.0 CACHE PATH "CANN")
set(OP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../csrc/operators/lightning_indexer_with_fetch_safe" CACHE PATH "Existing LI Fetch")
set(KVAF_INCLUDE "" CACHE PATH "KVAF device headers")
if(NOT KVAF_INCLUDE)
  message(FATAL_ERROR "Pass KVAF_INCLUDE to the patched batch-get device headers")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set(PARENT "${CMAKE_CURRENT_SOURCE_DIR}/..")
ascendc_library(serial_resolver STATIC serial_resolver.cpp)
ascendc_include_directories(serial_resolver PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${PARENT} ${OP_ROOT}/include ${OP_ROOT}/op_kernel ${KVAF_INCLUDE} ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(serial_resolver PRIVATE -O2 --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232 -DV9_SIMULATION=0 -DV9_BATCH_POST=2 -DSIM_CAPTURE_HITS=1)
ascendc_library(serial_update STATIC ${OP_ROOT}/op_kernel/index_cache_update.cpp)
ascendc_include_directories(serial_update PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(serial_update PRIVATE -O2 -mllvm -user-reserved-ub-size=223232)
ascendc_library(pipeline_parallel_kernel STATIC pipeline_kernel_parallel.cpp)
ascendc_include_directories(pipeline_parallel_kernel PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${PARENT} ${OP_ROOT}/include ${OP_ROOT}/op_kernel ${KVAF_INCLUDE} ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(pipeline_parallel_kernel PRIVATE -O2 --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232 -DV9_SIMULATION=0 -DV9_BATCH_POST=2 -DSIM_CAPTURE_HITS=1)
add_executable(pipeline_ab_parallel_sim pipeline_ab_sim.cpp ${OP_ROOT}/op_host/index_cache_update.cpp)
target_compile_definitions(pipeline_ab_parallel_sim PRIVATE PIPELINE_PARALLEL=1)
target_include_directories(pipeline_ab_parallel_sim PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OP_ROOT}/include ${KVAF_INCLUDE})
target_link_libraries(pipeline_ab_parallel_sim PRIVATE pipeline_parallel_kernel serial_resolver serial_update host_intf_pub ascendcl)
target_link_options(pipeline_ab_parallel_sim PRIVATE -Wl,--allow-shlib-undefined)
