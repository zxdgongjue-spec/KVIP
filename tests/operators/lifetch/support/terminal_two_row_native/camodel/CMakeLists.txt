cmake_minimum_required(VERSION 3.16)
project(two_row_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(JOINT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
execute_process(COMMAND ${Python3_EXECUTABLE} ${JOINT}/prepare_overlay.py ${OP}/op_kernel/terminal ${CMAKE_CURRENT_BINARY_DIR} RESULT_VARIABLE prep)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/overlay.py ${JOINT} ${CMAKE_CURRENT_BINARY_DIR} RESULT_VARIABLE hooks)
if(prep OR hooks)
    message(FATAL_ERROR "Source hooks do not match")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(two_row_sim_kernel STATIC ${CMAKE_CURRENT_BINARY_DIR}/kernel.cpp)
ascendc_include_directories(two_row_sim_kernel PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR} ${JOINT} ${OP}/include ${OP}/op_kernel ${OP}/op_kernel/indexer ${OP}/op_kernel/terminal ${KVAF_INCLUDE} ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(two_row_sim_kernel PRIVATE -DKVIP_PARALLEL_TERMINAL=1 -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232 -DV9_BATCH_POST=2 -DSIM_Q=4 -DSIM_CAPTURE_HITS=1 -DV9_SIMULATION=0 -DKVIP_STREAMING_LI=1 -DKVIP_OUTPUT_WORKER=1 -DROTATION_PUBLISH_TILES=1)
target_link_directories(two_row_sim_kernel PUBLIC ${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib)
add_executable(probe main.cpp)
target_include_directories(probe PRIVATE ${OP}/include ${OP}/op_kernel ${OP}/op_kernel/terminal ${KVAF_INCLUDE})
target_compile_definitions(probe PRIVATE KVIP_OUTPUT_WORKER=1)
target_link_libraries(probe PRIVATE two_row_sim_kernel)
target_link_options(probe PRIVATE "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/lib64" "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib")
