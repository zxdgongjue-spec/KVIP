cmake_minimum_required(VERSION 3.16)
project(window_rank_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(RUN_MODE sim CACHE STRING "")
set(SOC_VERSION Ascend950DT_9582 CACHE STRING "")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann-9.2.0 CACHE PATH "")
include(${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(window_rank STATIC kernel.cpp)
if(NOT INDEXER_DIR)
    set(INDEXER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../csrc/operators/lightning_indexer_with_fetch/op_kernel/indexer")
endif()
ascendc_include_directories(window_rank PRIVATE ${INDEXER_DIR})
ascendc_compile_options(window_rank PRIVATE -O2 --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232)
if(BASELINE_ONLY)
    ascendc_compile_definitions(window_rank PRIVATE BASELINE_ONLY=1)
endif()
target_link_directories(window_rank PUBLIC ${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib)
add_executable(probe main.cpp)
target_link_libraries(probe PRIVATE window_rank)
target_link_options(probe PRIVATE "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/lib64" "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib")

if(BUILD_HANDOFF)
    ascendc_library(window_handoff STATIC handoff.cpp)
    ascendc_include_directories(window_handoff PRIVATE ${INDEXER_DIR})
    ascendc_compile_options(window_handoff PRIVATE -O2 --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DV9_USER_UB_BYTES=223232)
    target_link_directories(window_handoff PUBLIC ${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib)
    add_executable(handoff handoff_main.cpp)
    target_link_libraries(handoff PRIVATE window_handoff)
    target_link_options(handoff PRIVATE "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/lib64" "-Wl,-rpath-link,${ASCEND_CANN_PACKAGE_PATH}/tools/simulator/dav_3510/lib")
endif()
