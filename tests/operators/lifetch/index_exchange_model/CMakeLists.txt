cmake_minimum_required(VERSION 3.16)
project(index_exchange_model LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(RUN_MODE sim)
set(SOC_VERSION Ascend950PR_9599 CACHE STRING "Simulator SoC")
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann CACHE PATH "CANN root")
get_filename_component(OP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../csrc/operators/lightning_indexer_with_fetch/op_kernel" ABSOLUTE)
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(index_exchange_kernel STATIC kernel.cpp)
ascendc_include_directories(index_exchange_kernel PRIVATE ${OP_ROOT} ${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/asc/include)
ascendc_compile_options(index_exchange_kernel PRIVATE -O2 -g --cce-auto-sync=off -mllvm -user-reserved-ub-size=223232 -DKVIP_PARALLEL_TERMINAL=1 -DKVIP_OUTPUT_WORKER=1)
add_executable(index_exchange_model driver.cpp)
target_link_libraries(index_exchange_model PRIVATE index_exchange_kernel host_intf_pub ascendcl)
# Simulator runtime resolves its model hooks when msprof loads the model.
target_link_options(index_exchange_model PRIVATE -Wl,--allow-shlib-undefined)
